# Simulated Weeds + OpenCV Weed Detection Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add oversized magenta weed blobs to the Gazebo farm worlds via a seeded generator, and extend `CameraProcessor` with HSV-based OpenCV weed detection whose georeferenced results land in the per-waypoint JSON sidecars.

**Architecture:** A generator script emits a single static visual-only Gazebo model (`weed_field`) with N weed links plus a `ground_truth.json`; both swarm worlds `<include>` it. Detection is a new private method on `CameraProcessor` (mirroring `_detect_aruco`) called from `capture()`, with nadir pinhole georeferencing.

**Tech Stack:** Python 3, OpenCV (cv2, HSV thresholding + contours), Gazebo SDF 1.9. Tests are plain-python scripts (pytest on this host is unreliable — run test files directly with `python3`).

**Spec:** `docs/superpowers/specs/2026-07-23-simulated-weeds-detection-design.md`

## Global Constraints

- Weeds are visual-only: **no collision geometry** anywhere in the weed model.
- Weed color: magenta `1 0 1 1` for ambient, diffuse, AND emissive.
- Weed diameter: uniform random in **1.2–2.5 m**; blob height 0.2 m, link pose z = 0.2.
- Placement area: **80×80 m square centered at Gazebo (45, 50)**; defaults `--count 12 --seed 42`.
- Field origin constants: `LAT_BASE=40.072842`, `LON_BASE=-105.230575` (must match `src/scripts/start_drone_container.sh`).
- HSV detection band: low `(140, 100, 80)`, high `(170, 255, 255)`; min contour area **50 px**.
- Camera intrinsics: `horizontal_fov` **1.2 rad**, 640×480 (from `src/custom_gz/models/base_drone_cam/model.sdf`) — read frame size from the frame, keep HFOV as a module constant.
- `CameraProcessor` non-fatal contract: any failure in detection returns `[]`; the mission must never break.
- Do not modify `src/scripts/stitch_field_map.py` (out of scope).
- Run tests with `python3 tests/<file>.py` (NOT pytest). Every test file ends with a `__main__` runner that executes each test function and prints `ALL PASS`.

```python
#!/usr/bin/env python3
"""Generate the weed_field Gazebo model: oversized magenta weed blobs.

Weeds are deliberately unrealistic — large flat magenta cylinders scattered
over the surveyed 80x80 m area — so the drones' OpenCV pipeline has an
unambiguous, tunable detection target (HSV band in camera_processor.py).

Output is a single static, visual-only model at src/custom_gz/models/weed_field
(included by swarm_drone.sdf and swarm_drone_spray.sdf) plus ground_truth.json
with each weed's Gazebo x/y and derived lat/lon for checking detections.

Regenerate with a different layout:
    python3 src/custom_gz/worlds/generate_weeds.py --count 20 --seed 7
"""
import argparse
import json
import math
import os
import random

# Must match LAT_BASE/LON_BASE in src/scripts/start_drone_container.sh and
# FIELD_ORIGIN_LAT/LON in compose_decentralized.launch.py. The field origin
# corresponds to Gazebo (45, 50) — drone 0's spawn/home.
LAT_BASE = 40.072842
LON_BASE = -105.230575
METERS_PER_DEG_LAT = 111111.0

DIAMETER_MIN = 1.2
DIAMETER_MAX = 2.5
BLOB_HEIGHT = 0.2
BLOB_Z = 0.2  # field surface tops out at z=0.1; blob center 0.2 -> spans 0.1..0.3


def gazebo_to_latlon(x, y, center_x, center_y):
    """Map Gazebo world x/y to lat/lon, with (center_x, center_y) = field origin."""
    north = y - center_y
    east = x - center_x
    lat = LAT_BASE + north / METERS_PER_DEG_LAT
    lon = LON_BASE + east / (METERS_PER_DEG_LAT * math.cos(math.radians(LAT_BASE)))
    return lat, lon


def generate_weeds(count, seed, center, area):
    rng = random.Random(seed)
    half = area / 2.0
    weeds = []
    for i in range(count):
        x = round(rng.uniform(center[0] - half, center[0] + half), 3)
        y = round(rng.uniform(center[1] - half, center[1] + half), 3)
        diameter = round(rng.uniform(DIAMETER_MIN, DIAMETER_MAX), 3)
        lat, lon = gazebo_to_latlon(x, y, center[0], center[1])
        weeds.append({
            "id": i, "x": x, "y": y, "diameter": diameter,
            "lat": round(lat, 8), "lon": round(lon, 8),
        })
    return weeds


def _weed_link(w):
    return f"""    <link name="weed_{w['id']}">
      <pose>{w['x']} {w['y']} {BLOB_Z} 0 0 0</pose>
      <visual name="visual">
        <geometry>
          <cylinder>
            <radius>{w['diameter'] / 2.0}</radius>
            <length>{BLOB_HEIGHT}</length>
          </cylinder>
        </geometry>
        <material>
          <ambient>1 0 1 1</ambient>
          <diffuse>1 0 1 1</diffuse>
          <emissive>1 0 1 1</emissive>
        </material>
      </visual>
    </link>
"""


def model_sdf(weeds):
    links = "".join(_weed_link(w) for w in weeds)
    return f"""<?xml version="1.0"?>
<sdf version="1.9">
  <!-- GENERATED by src/custom_gz/worlds/generate_weeds.py - do not hand-edit.
       Oversized magenta weed blobs for OpenCV detection testing. Visual-only
       (no collision) so they can never interfere with flight. Ground truth
       positions: ground_truth.json next to this file. -->
  <model name="weed_field">
    <static>true</static>
{links}  </model>
</sdf>
"""


MODEL_CONFIG = """<?xml version="1.0"?>
<model>
  <name>weed_field</name>
  <version>1.0</version>
  <sdf version="1.9">model.sdf</sdf>
  <author>
    <name>Yahia Aly</name>
    <email>yahia@ezrpa.com</email>
  </author>
  <description>
    Generated magenta weed blobs over the surveyed field area - detection
    targets for the drones' OpenCV camera pipeline. See generate_weeds.py.
  </description>
</model>
"""


def write_model(weeds, out_dir, meta):
    os.makedirs(out_dir, exist_ok=True)
    with open(os.path.join(out_dir, "model.sdf"), "w") as f:
        f.write(model_sdf(weeds))
    with open(os.path.join(out_dir, "model.config"), "w") as f:
        f.write(MODEL_CONFIG)
    with open(os.path.join(out_dir, "ground_truth.json"), "w") as f:
        json.dump({**meta, "weeds": weeds}, f, indent=2)


def main():
    default_out = os.path.normpath(os.path.join(
        os.path.dirname(os.path.abspath(__file__)), "..", "models", "weed_field"))
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--count", type=int, default=12)
    p.add_argument("--seed", type=int, default=42)
    p.add_argument("--center", type=float, nargs=2, default=[45.0, 50.0],
                   metavar=("X", "Y"), help="Gazebo x/y of the field origin")
    p.add_argument("--area", type=float, default=80.0, help="side of the square area (m)")
    p.add_argument("--out", default=default_out)
    args = p.parse_args()

    weeds = generate_weeds(args.count, args.seed, tuple(args.center), args.area)
    write_model(weeds, args.out, {
        "seed": args.seed, "count": args.count,
        "center": args.center, "area": args.area,
    })
    print(f"wrote {len(weeds)} weeds to {args.out}")


if __name__ == "__main__":
    main()
```

```python
    def _detect_weeds(self, frame, lat=None, lon=None, alt=None):
        """Detect magenta weed blobs; returns [{'centroid_px', 'bbox',
        'area_px', 'lat', 'lon'}]. lat/lon are georeferenced from the drone
        pose assuming a north-aligned nadir camera (approximate — vehicle yaw
        is not compensated), or None when the pose is unknown."""
        try:
            hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)
            mask = cv2.inRange(hsv, WEED_HSV_LOW, WEED_HSV_HIGH)
            kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
            mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
            mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
            contours, _ = cv2.findContours(
                mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

            h, w = frame.shape[:2]
            f_px = (w / 2.0) / math.tan(CAMERA_HFOV_RAD / 2.0)
            detections = []
            for c in contours:
                area = cv2.contourArea(c)
                if area < WEED_MIN_AREA_PX:
                    continue
                bx, by, bw, bh = cv2.boundingRect(c)
                cx = bx + bw / 2.0
                cy = by + bh / 2.0
                det = {
                    "centroid_px": [cx, cy],
                    "bbox": [bx, by, bw, bh],
                    "area_px": area,
                    "lat": None,
                    "lon": None,
                }
                if lat is not None and lon is not None and alt is not None:
                    east = (cx - w / 2.0) / f_px * alt
                    north = -(cy - h / 2.0) / f_px * alt
                    det["lat"] = lat + north / 111111.0
                    det["lon"] = lon + east / (
                        111111.0 * math.cos(math.radians(lat)))
                detections.append(det)
            return detections
        except Exception:
            return []
```

- [ ] **Step 4: Check sidecars against ground truth**

```bash
python3 - <<'EOF'
import glob, json, math
gt = json.load(open("src/custom_gz/models/weed_field/ground_truth.json"))["weeds"]
dets = []
for f in sorted(glob.glob("logs/drone_*/wp_*.json")):
    dets += [d for d in json.load(open(f)).get("weed_detections", []) if d["lat"]]
print(f"{len(dets)} georeferenced detections across sidecars")
for d in dets[:20]:
    best = min(gt, key=lambda w: (w["lat"]-d["lat"])**2 + (w["lon"]-d["lon"])**2)
    err = math.hypot((best["lat"]-d["lat"])*111111,
                     (best["lon"]-d["lon"])*111111*math.cos(math.radians(d["lat"])))
    print(f"  det@({d['lat']:.6f},{d['lon']:.6f}) -> weed {best['id']} err {err:.1f} m")
EOF
```
Expected: detections present, typical error a few meters (georeferencing is approximate; < ~10 m is fine at 30 m altitude). Then `./stop_swarm.sh` and kill the Gazebo process.
